cmake_minimum_required(VERSION 3.20)
project(tab_search LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()

include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json
    URL https://github.com/nlohmann/json/archive/refs/tags/v3.12.0.tar.gz
    URL_HASH SHA256=4b92eb0c06d10683f7447ce9406cb97cd4b453be18d7279320f7b2f025c10187)
set(LIBXML2_WITH_PROGRAMS OFF CACHE BOOL "" FORCE)
set(LIBXML2_WITH_TESTS OFF CACHE BOOL "" FORCE)
set(LIBXML2_WITH_PYTHON OFF CACHE BOOL "" FORCE)
set(LIBXML2_WITH_ICONV OFF CACHE BOOL "" FORCE)
set(LIBXML2_WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(LIBXML2_WITH_LZMA OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(libxml2
    URL https://github.com/GNOME/libxml2/archive/refs/tags/v2.14.6.tar.gz
    URL_HASH SHA256=546ab74561c040df210c88dbd3c652bf509d826954ab2002c8973f1fa8d10130)
FetchContent_MakeAvailable(json libxml2)

add_library(tab_chords chords.cpp)
target_include_directories(tab_chords PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(tab_providers providers.cpp http.cpp)
target_include_directories(tab_providers PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tab_providers PUBLIC tab_chords PRIVATE nlohmann_json::nlohmann_json LibXml2::LibXml2)
if(WIN32)
    target_link_libraries(tab_providers PRIVATE winhttp)
else()
    find_package(CURL REQUIRED)
    target_link_libraries(tab_providers PRIVATE CURL::libcurl)
endif()

add_executable(tab_search main.cpp)
add_library(tab_musicxml musicxml.cpp)
target_link_libraries(tab_musicxml PUBLIC tab_chords)
target_include_directories(tab_musicxml PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tab_search PRIVATE tab_providers tab_musicxml)

option(BUILD_GUI "Build the Qt Widgets desktop application" ON)
if(BUILD_GUI)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Concurrent)
    add_library(tab_gui gui/main_window.cpp)
    target_include_directories(tab_gui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(tab_gui PUBLIC tab_providers tab_musicxml Qt6::Widgets Qt6::Concurrent)
    add_executable(tab_search_gui WIN32 gui/main.cpp)
    target_link_libraries(tab_search_gui PRIVATE tab_gui)
    get_target_property(qmake_path Qt6::qmake IMPORTED_LOCATION)
    if(WIN32)
        get_filename_component(qt_bin "${qmake_path}" DIRECTORY)
        find_program(WINDEPLOYQT_EXECUTABLE windeployqt HINTS "${qt_bin}" REQUIRED)
        add_custom_command(TARGET tab_search_gui POST_BUILD
            COMMAND "${WINDEPLOYQT_EXECUTABLE}" --no-translations --no-compiler-runtime
                --no-system-d3d-compiler --no-opengl-sw "$<TARGET_FILE:tab_search_gui>"
            COMMENT "Deploying Qt runtime for the desktop application"
            VERBATIM)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(chord_tests tests/chord_tests.cpp)
    target_link_libraries(chord_tests PRIVATE tab_musicxml LibXml2::LibXml2)
    add_test(NAME chords COMMAND chord_tests)
    add_executable(provider_tests tests/provider_tests.cpp)
    target_link_libraries(provider_tests PRIVATE tab_providers)
    add_test(NAME providers COMMAND provider_tests)
    add_executable(console_tests tests/console_tests.cpp)
    target_link_libraries(console_tests PRIVATE tab_musicxml)
    add_test(NAME console_export COMMAND console_tests)
    if(BUILD_GUI)
        find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
        add_executable(gui_tests tests/gui_tests.cpp)
        set_target_properties(gui_tests PROPERTIES AUTOMOC ON)
        target_link_libraries(gui_tests PRIVATE tab_gui Qt6::Test)
        add_test(NAME gui COMMAND gui_tests -o -,txt)
        execute_process(COMMAND "${qmake_path}" -query QT_INSTALL_PLUGINS
            OUTPUT_VARIABLE qt_plugins OUTPUT_STRIP_TRAILING_WHITESPACE
            COMMAND_ERROR_IS_FATAL ANY)
        file(TO_CMAKE_PATH "${qt_plugins}" qt_plugins)
        set_tests_properties(gui PROPERTIES ENVIRONMENT
            "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORM_PLUGIN_PATH=${qt_plugins}/platforms")
        if(WIN32)
            set_tests_properties(gui PROPERTIES
                ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${qt_bin}")
        endif()
    endif()
endif()

set(app_targets tab_chords tab_providers tab_musicxml tab_search)
if(BUILD_GUI)
    list(APPEND app_targets tab_gui tab_search_gui)
endif()
foreach(target IN LISTS app_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
